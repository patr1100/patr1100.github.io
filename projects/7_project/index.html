              def get_page(url:str) -> Union[str, list]:
                  def tag_visible(element):
                      if element.parent.name in ['style', 'script', 'head', 'title', 'meta', '[document]']:
                          return False
                      if isinstance(element, Comment):
                          return False
                      return True
                  rand_user_agents = ['Mozilla/5.0 (Windows NT 10.0; Win64; x64; rv:126.0) Gecko/20100101 Firefox/126.1', 'Mozilla/5.0 (Windows NT 10.0; Win64; x64; rv:109.0) Gecko/20100101 Firefox/118.0', 'Mozilla/5.0 (Macintosh; Intel Mac OS X 10.15; rv:127.0) Gecko/20100101 Firefox/127.0']
                  agent = rand_user_agents[rnd.randint(0, len(rand_user_agents)-1)]
                  try:
                      # Configure logging
                      logging.basicConfig(level=logging.INFO)
                      logger = logging.getLogger(__name__)
              
                      # Create a session
                      session = requests.Session()
              
                      retry_strategy = Retry(
                          total=5,
                          backoff_factor=1,
                          status_forcelist=[429, 500, 502, 503, 504],
                          allowed_methods=["HEAD", "GET", "OPTIONS"]
                      )
              
                      # Mount the retry strategy to the session
                      adapter = HTTPAdapter(max_retries=retry_strategy)
                      session.mount("http://", adapter)
                      session.mount("https://", adapter)
              
                      response = session.get(url, headers={'User-Agent':agent}, timeout=120)
                  except requests.exceptions.Timeout:
                      time = f"Request took too long skipping link: {url}"
                      echo(time)
                      logger.error(time)
                      return("", list()) # I don't have all day to wait around for one web page
                  except requests.exceptions.ConnectionError as e:
                      logger.error(f"Connection error: {e}")
                      return("", list())
                  except requests.exceptions.HTTPError as e:
                      logger.error(f"HTTP error: {e}")
                      return("", list())
                  except requests.exceptions.RequestException as e:
                      logger.error(f"General error: {e}")
                      return("", list())
                  try:
                      doc = Document(response.text)
                      doc_dom = BeautifulSoup(doc.summary(), features='lxml')
                      # print(doc_dom)
                      texts = doc_dom.find_all(text=True)
                  except readability.Unparseable:
                      echo(f"Could not parse the document skipping link: {url}")
                      return("", list()) # The website could get meaningful text from the website.
              
                  visible_texts = filter(tag_visible, texts)
                  cleanish_text = u" ".join(t.strip() for t in visible_texts)
                  clean_text = unicodedata.normalize('NFKD', cleanish_text).encode('ascii', 'ignore')
                      
                  # print(clean_text)
                  a_list = list() # These links are very important portals
                  
                  for a in doc_dom.find_all('a'):
                      try:
                          a_url = urllib.parse.urlparse(a['href'])
                          if not a_url.hostname:
                              a['href'] = urllib.parse.urljoin(url, a['href'])
                          a_list.append(a['href'])
                      except:
                          echo('Malformed a tag - no href - skipping: ' + str(a))
                          continue
                  
                  return(clean_text.decode('ascii'), a_list)
              
              
              def create_date_folders(dt: datetime.datetime):
                  # Extract year, month, and day from the datetime object
                  year = dt.year
                  month = dt.month
                  day = dt.day
              
                  # Define the base directory where the folders will be created
                  base_dir = os.path.join(os.path.dirname(os.getcwd()), 'data', 'news') # This is assuming that this program is being ran from the programs folder
              
                  # Create the folder structure
                  year_folder = os.path.join(base_dir, str(year))
                  month_folder = os.path.join(year_folder, f"{month:02d}")  # Zero-padded month
                  day_folder = os.path.join(month_folder, f"{day:02d}")      # Zero-padded day
              
                  for f in [year_folder, month_folder, day_folder]:
                      os.makedirs(f, exist_ok=True)

              def get_stock_name(data: dict, name_list: list, symbol: str):
                  def most_common_term(lst):
                      if not lst:  # Check if the list is empty
                          return None
                      
                      # Create a Counter object to count occurrences of each term
                      counter = Counter(lst)
                      
                      # Find the most common term and its count
                      most_common = counter.most_common(1)  # Returns a list of the n most common elements and their counts
                      
                      if most_common:
                          return most_common[0][0]  # Return the term (the first element of the tuple)
                      return None
                  first_info = list(data['info'].keys())[0]
                  info_keys = data['info'][first_info].keys()
                  for n in name_list:
                      if n in info_keys:
                          return data['info'][first_info][n]# Return the first encounted instance of the stock name in the  first info entry
                  if "extractedNames" in data.keys():
                      ex_names = data["extractedNames"]
                      return(most_common_term(ex_names))
                  else:
                      return(symbol)# No alias was found return just the ticker
